<!-- Mount-Punkt für den geladenen Content -->
<div id="bsrp-mount" style="background:#faf8f5;min-height:80vh"></div>

<script>
  /* =============================================================
     LOADER · Performance-optimiert
     -----------------------------------------------------------------
     Ladekette für sofortige Sichtbarkeit:
       1) Resource Hints sofort in <head> injizieren
          - preconnect zu Webflow-CDN (TLS-Handshake parallel)
          - dns-prefetch zu YouTube (für Video-Facade später)
          - preload retreat-content.txt (parallel zum HTML-Parse)
          - preload Hero-Banner-Bild (fetchpriority=high)
       2) Skeleton sofort im Mount (kein 700ms Spinner-Delay)
       3) fetch + DocumentFragment-Inject (1 Reflow statt mehrerer)
       4) <script>-Tags manuell re-aktivieren (Hot-Swap im Content)
     ============================================================= */
  (function(){
    var mount = document.getElementById('bsrp-mount');
    if (!mount) return;

    var cfg = window.BSRP_CONFIG || {};
    var CONTENT_URL = cfg.contentUrl;
    var CONFIGURED  = CONTENT_URL && CONTENT_URL.indexOf('CONTENT_URL_HIER') !== 0;

    /* --- 1) Resource Hints sofort ----------------------------- */
    var head = document.head || document.getElementsByTagName('head')[0];
    function hint(rel, href, attrs){
      if (!head || !href) return;
      var l = document.createElement('link');
      l.rel  = rel;
      l.href = href;
      if (attrs) for (var k in attrs) l.setAttribute(k, attrs[k]);
      head.appendChild(l);
    }
    hint('preconnect',   'https://cdn.prod.website-files.com', { crossorigin: 'anonymous' });
    hint('preconnect',   'https://raw.githubusercontent.com',  { crossorigin: 'anonymous' });
    hint('dns-prefetch', 'https://i.ytimg.com');
    hint('dns-prefetch', 'https://www.youtube.com');
    if (CONFIGURED) {
      hint('preload', CONTENT_URL, { as: 'fetch', crossorigin: 'anonymous' });
    }
    if (cfg.images && cfg.images.heroBanner) {
      hint('preload', cfg.images.heroBanner, { as: 'image', fetchpriority: 'high' });
    }
    if (cfg.images && cfg.images.logo) {
      hint('preload', cfg.images.logo, { as: 'image', fetchpriority: 'high' });
    }

    /* --- 2) Skeleton sofort, kein Spinner-Delay --------------- */
    if (!CONFIGURED) {
      mount.innerHTML =
        '<div style="padding:1.5rem;border:1px solid #d9534f;background:#fff5f5;color:#7b2020;border-radius:8px;font-family:system-ui,sans-serif;font-size:14px;line-height:1.5">' +
          '<strong>Retreat-LP nicht konfiguriert.</strong><br>' +
          'Bitte <code>BSRP_CONFIG.contentUrl</code> mit der URL der retreat-content.txt füllen.' +
        '</div>';
      return;
    }
    mount.innerHTML =
      '<div style="min-height:80vh;background:#faf8f5;display:flex;align-items:center;justify-content:center;font-family:Manrope,system-ui,-apple-system,Segoe UI,Roboto,sans-serif;">' +
        '<span aria-label="Inhalte werden geladen" style="display:inline-block;width:28px;height:28px;border:3px solid rgba(74,112,140,0.18);border-top-color:#4a708c;border-radius:50%;animation:bsrp-spin 0.7s linear infinite;"></span>' +
      '</div>' +
      '<style>@keyframes bsrp-spin{to{transform:rotate(360deg)}}</style>';

    /* --- Fehler-Helper --------------------------------------- */
    function showError(msg, detail) {
      mount.innerHTML =
        '<div style="padding:2rem 1.5rem;text-align:center;color:#7b2020;background:#fff5f5;border:1px solid #d9534f;border-radius:8px;font-family:system-ui,-apple-system,Segoe UI,Roboto,sans-serif;font-size:14px;line-height:1.5">' +
          '<strong style="font-size:15px">Retreat-LP konnte nicht geladen werden.</strong><br>' +
          '<span style="opacity:0.8">' + msg + '</span>' +
          (detail ? '<br><small style="opacity:0.6;font-family:monospace;font-size:12px">' + detail + '</small>' : '') +
        '</div>';
    }
    if (!window.fetch) { showError('Browser unterstützt fetch() nicht.', ''); return; }

    /* --- 3) Fetch + 4) Inject --------------------------------- */
    fetch(CONTENT_URL, { credentials: 'omit' })
      .then(function(r){
        if (!r.ok) throw new Error('HTTP ' + r.status + ' ' + r.statusText);
        return r.text();
      })
      .then(function(html){
        if (!html || html.length < 100) throw new Error('Leere Antwort');

        /* DocumentFragment-Inject statt innerHTML: weniger Reflows,
           Parsing parallel. Skripte werden allerdings nicht
           automatisch ausgeführt - das machen wir gleich. */
        mount.innerHTML = '';
        var range = document.createRange();
        range.selectNode(mount);
        var fragment = range.createContextualFragment(html);
        mount.appendChild(fragment);

        /* Skript-Tags reaktivieren, damit das Hot-Swap im Content
           gegen BSRP_CONFIG läuft. */
        var scripts = mount.querySelectorAll('script');
        scripts.forEach(function(oldScript){
          var newScript = document.createElement('script');
          for (var i = 0; i < oldScript.attributes.length; i++) {
            var a = oldScript.attributes[i];
            newScript.setAttribute(a.name, a.value);
          }
          newScript.textContent = oldScript.textContent;
          oldScript.parentNode.replaceChild(newScript, oldScript);
        });
      })
      .catch(function(err){
        console.error('[Retreat-LP] Content-Load fehlgeschlagen:', err);
        showError(err.message || 'Unbekannter Fehler', CONTENT_URL);
      });
  })();
</script>
