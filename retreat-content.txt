    /* 1) Bild-URLs aus cfg.images übernehmen */
    if (cfg.images) {
      document.querySelectorAll('[data-bsrp-img]').forEach(function(el){
        var key = el.getAttribute('data-bsrp-img');
        if (cfg.images[key]) el.src = cfg.images[key];
      });
    }

    /* 2) Link-URLs aus cfg.urls übernehmen */
    if (cfg.urls) {
      document.querySelectorAll('[data-bsrp-url]').forEach(function(el){
        var key = el.getAttribute('data-bsrp-url');
        if (cfg.urls[key]) el.href = cfg.urls[key];
      });
    }

    /* 3) Termine aus cfg.termine rendern (mit Fallback bei leerer Liste) */
    var mount = document.getElementById('bsrp-termine-mount');
    if (mount) {
      if (Array.isArray(cfg.termine) && cfg.termine.length) {
        mount.innerHTML = cfg.termine.map(function(t){
          var cls   = 'bsrp-termin' + (t.upcoming ? ' upcoming' : '');
          var tag   = t.tag ? '<span class="bsrp-termin-tag">' + esc(t.tag) + '</span>' : '';
          return ''
            + '<article class="' + cls + '">'
            +   '<div>'
            +     '<div class="bsrp-termin-main">' + esc(t.title) + '</div>'
            +     (t.sub ? '<div class="bsrp-termin-sub">' + esc(t.sub) + '</div>' : '')
            +   '</div>'
            +   tag
            + '</article>';
        }).join('');
      } else {
        /* Leeres Termine-Array: freundlicher Hinweis statt leerer Sektion */
        mount.innerHTML =
          '<article class="bsrp-termin">' +
            '<div>' +
              '<div class="bsrp-termin-main">Neue Termine in Planung</div>' +
              '<div class="bsrp-termin-sub">Schreib uns über die Beratung, sobald ein Datum steht.</div>' +
            '</div>' +
            '<span class="bsrp-termin-tag">Bald</span>' +
          '</article>';
      }
    }
  })();
</script>


<!-- =====================================================================
     JS · Testimonial-Videos: Click-to-Play Facade
     Spart Bandbreite und versteckt YouTube-Branding bis zum Klick.
     Beim Klick wird die Thumbnail-Button durch das iframe ersetzt
     (mit autoplay=1, damit das Video sofort startet).
     ===================================================================== -->
<script>
  (function(){
    function buildIframe(id, label){
      var iframe = document.createElement("iframe");
      iframe.src = "https://www.youtube.com/embed/" + id +
                   "?autoplay=1&rel=0&modestbranding=1&playsinline=1";
      iframe.title = label || "Video-Testimonial";
      iframe.setAttribute("allow",
        "accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share; fullscreen");
      iframe.setAttribute("allowfullscreen", "");
      iframe.setAttribute("frameborder", "0");
      return iframe;
    }

    document.querySelectorAll(".bsrp-testi-play").forEach(function(btn){
      /* Thumbnail-Fallback: maxresdefault → hqdefault (manche Videos
         haben kein maxres, z.B. Shorts; dann auf 480x360 zurückfallen) */
      var img = btn.querySelector("img");
      if (img) {
        img.addEventListener("error", function once(){
          img.removeEventListener("error", once);
          img.src = img.src.replace("maxresdefault", "hqdefault");
        }, { once: true });
      }

      btn.addEventListener("click", function(e){
        e.preventDefault();
        var id = btn.getAttribute("data-bsrp-yt");
        if (!id) return;
        var label = btn.getAttribute("aria-label");
        var iframe = buildIframe(id, label);
        var wrap = btn.parentNode;
        wrap.replaceChild(iframe, btn);
      });
    });
  })();
</script>


<!-- =====================================================================
     JS · Warteliste-Formular (Validierung, Webhook, Redirect-Routing)
     ===================================================================== -->
<script>
  (function(){
    "use strict";

    /* -----------------------------------------------------------------
       KONFIG  -  URLs split-string-obfuskiert (analog zur Webinar-LP).
       Redirect-URLs werden aus BSRP_CONFIG.urls überschrieben, falls
       gesetzt (siehe retreat-embed-loader.html).
    ----------------------------------------------------------------- */
    var _cfg = {
      endpoint: [
        "https://hook.eu2",
        ".make.com/",
        "uut4cdx4hqs8imkk95jm2lek3qlb1mvh"
      ].join(""),
      trackingEndpoint: "https://barbara-scharrer-tracking.vercel.app/api/lead",
      redirectCustomer:    "https://www.barbarascharrer.de/retreat-warteliste",
      redirectNonCustomer: "https://www.barbarascharrer.de/retreat-beratung",
      minFillTime: 3000,
      submitTimeoutMs: 8000
    };

    var _err = {
      kundenstatus: "Bitte wähle, ob du bereits im Programm bist.",
      first_name:   "Bitte gib deinen Vornamen ein.",
      last_name:    "Bitte gib deinen Nachnamen ein.",
      email:        "Bitte gib eine gültige E-Mail-Adresse ein.",
      phone:        "Bitte gib eine gültige Telefonnummer ein.",
      datenschutz:  "Bitte stimme der Datenschutzerklärung zu."
    };

    function $id(id){ return document.getElementById(id); }

    function setFieldError(field, errId, msg){
      var e = $id(errId);
      if (e) e.textContent = msg || "";
      if (field) field.classList.toggle("is-invalid", !!msg);
    }

    function clearAllErrors(form){
      form.querySelectorAll(".field-error").forEach(function(n){ n.textContent = ""; });
      form.querySelectorAll(".is-invalid").forEach(function(n){ n.classList.remove("is-invalid"); });
      var gErr = $id("form-error-w");
      if (gErr) { gErr.textContent = ""; gErr.style.display = "none"; }
    }

    function showGlobalError(msg){
      var gErr = $id("form-error-w");
      if (!gErr) return;
      gErr.textContent = msg;
      gErr.style.display = "block";
    }

    function isValidEmail(s){
      return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(s);
    }
    function isValidPhone(s){
      var digits = s.replace(/\D+/g, "");
      if (digits.length < 6 || digits.length > 15) return false;
      /* Trivial-Sequenzen abfangen: 123456, 000000, etc. */
      if (/^(.)\1+$/.test(digits)) return false;
      if (/^012345|^123456|^234567/.test(digits)) return false;
      return true;
    }

    function fillUtmHidden(form){
      var p = new URLSearchParams(window.location.search);
      ["utm_source","utm_medium","utm_campaign","utm_term","utm_content"].forEach(function(n){
        var v = p.get(n);
        if (v) {
          var el = form.querySelector("[name='" + n + "']");
          if (el) el.value = v;
        }
      });
    }

    function collectData(form){
      var data = {};
      var fd = new FormData(form);
      fd.forEach(function(v, k){
        /* Mehrfach-Felder zusammenfügen (kommt hier nicht vor, aber safe) */
        if (data[k] !== undefined) {
          if (Array.isArray(data[k])) data[k].push(v);
          else data[k] = [data[k], v];
        } else {
          data[k] = v;
        }
      });
      /* phone = prefix + number */
      var prefix = (form.querySelector("[name='phone_prefix']") || {}).value || "";
      if (data.phone && prefix) {
        data.phone_e164 = (prefix + " " + String(data.phone).replace(/^0+/, "").trim()).replace(/\s+/g, " ").trim();
      }
      /* Honeypot raus, ist nur Bot-Spüre */
      delete data.website;
      /* Form-Start-Time → Bot-Indikator separat hinten */
      data._meta = {
        page: window.location.pathname,
        url:  window.location.href,
        referrer: document.referrer || null,
        ts: new Date().toISOString()
      };
      return data;
    }

    function resolveRedirect(form){
      var urls = (window.BSRP_CONFIG && window.BSRP_CONFIG.urls) || {};
      var status = form.querySelector("input[name='kundenstatus']:checked");
      var isCustomer = status && status.value === "ja";
      return isCustomer
        ? (urls.warteliste      || _cfg.redirectCustomer)
        : (urls.retreatBeratung || _cfg.redirectNonCustomer);
    }

    function postJSON(url, body, timeoutMs){
      if (!url) return Promise.resolve(null);
      var ctrl = (typeof AbortController === "function") ? new AbortController() : null;
      var to   = ctrl ? setTimeout(function(){ ctrl.abort(); }, timeoutMs || 8000) : null;
      return fetch(url, {
        method: "POST",
        mode: "cors",
        credentials: "omit",
        keepalive: true,
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(body),
        signal: ctrl ? ctrl.signal : undefined
      }).finally(function(){ if (to) clearTimeout(to); });
    }

    function validateForm(form){
      clearAllErrors(form);
      var ok = true;

      /* Kundenstatus */
      var status = form.querySelector("input[name='kundenstatus']:checked");
      if (!status) { setFieldError(null, "err_kundenstatus", _err.kundenstatus); ok = false; }

      /* Vorname */
      var fn = form.querySelector("[name='first_name']");
      if (!fn.value.trim()) { setFieldError(fn, "err_first_name", _err.first_name); ok = false; }

      /* Nachname */
      var ln = form.querySelector("[name='last_name']");
      if (!ln.value.trim()) { setFieldError(ln, "err_last_name", _err.last_name); ok = false; }

      /* E-Mail */
      var em = form.querySelector("[name='email']");
      if (!isValidEmail(em.value.trim())) { setFieldError(em, "err_email", _err.email); ok = false; }

      /* Telefon */
      var ph = form.querySelector("[name='phone']");
      if (!isValidPhone(ph.value.trim())) { setFieldError(ph, "err_phone", _err.phone); ok = false; }

      /* Datenschutz */
      var ds = form.querySelector("[name='datenschutz']");
      if (!ds.checked) { setFieldError(null, "err_datenschutz", _err.datenschutz); ok = false; }

      return ok;
    }

    function initForm(){
      var form = $id("form-warteliste");
      if (!form) return;

      /* Start-Time setzen */
      var st = form.querySelector("[name='formStartTime']");
      var startMs = Date.now();
      if (st) st.value = String(startMs);

      /* UTM aus URL ziehen */
      fillUtmHidden(form);

      /* Felder live entglühen, sobald Eingabe da ist */
      form.addEventListener("input", function(e){
        var t = e.target;
        if (!t || !t.classList) return;
        if (t.classList.contains("is-invalid")) t.classList.remove("is-invalid");
        var wrap = t.closest(".form_field-wrap");
        if (wrap) {
          var err = wrap.querySelector(".field-error");
          if (err) err.textContent = "";
        }
      });

      /* Submit */
      form.addEventListener("submit", function(e){
        e.preventDefault();

        /* Honeypot: wenn ausgefüllt → still scheitern */
        var hp = form.querySelector("[name='website']");
        if (hp && hp.value) {
          window.location.href = resolveRedirect(form);
          return;
        }

        /* Bot-Indikator: zu schnell ausgefüllt */
        if (Date.now() - startMs < _cfg.minFillTime) {
          showGlobalError("Bitte nimm dir kurz Zeit, das Formular sorgfältig auszufüllen.");
          return;
        }

        if (!validateForm(form)) {
          showGlobalError("Bitte prüfe die markierten Felder.");
          return;
        }

        var btn = $id("form-submit-btn-w");
        var origLabel = btn ? btn.textContent : "";
        if (btn) {
          btn.disabled = true;
          btn.textContent = "Wird gesendet…";
        }

        var data     = collectData(form);
        var redirect = resolveRedirect(form);

        /* Parallel: Hauptwebhook + Tracking (Tracking-Fehler ignorieren) */
        var pMain  = postJSON(_cfg.endpoint,         data, _cfg.submitTimeoutMs);
        var pTrack = postJSON(_cfg.trackingEndpoint, data, _cfg.submitTimeoutMs).catch(function(){ return null; });

        Promise.all([pMain, pTrack])
          .then(function(){ window.location.href = redirect; })
          .catch(function(err){
            console.error("[BSRP] Warteliste-Submit fehlgeschlagen:", err);
            if (btn) { btn.disabled = false; btn.textContent = origLabel; }
            showGlobalError("Da ist gerade etwas schiefgegangen. Bitte versuche es nochmal, oder schreib uns direkt: info@barbarascharrer.de");
          });
      });
    }

    if (document.readyState === "loading") {
      document.addEventListener("DOMContentLoaded", initForm);
    } else {
      initForm();
    }
  })();
</script>
