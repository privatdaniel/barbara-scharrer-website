        <p>Das liegt selten an schlechten Ärzten. Es liegt daran, dass die Schulmedizin vor allem darauf ausgelegt ist, Symptome zu behandeln und zu lindern. Das funktioniert gut bei akuten Erkrankungen. Bei chronischen Beschwerden wie Inkontinenz braucht es aber mehr: <strong>Wird nur das Symptom behandelt und nicht die Ursache, kommt es immer wieder.</strong> Ein paar Einheiten Physiotherapie, eine Broschüre mit Übungen. Danach bist du wieder auf dich gestellt. Und ein Termin im Beckenbodenzentrum dauert gern acht Monate.</p>

        <p>Die Produkte, auf die du dann angewiesen bist, sind genau dafür gemacht: Hilfsmittel und Geräte, die dir überall begegnen. Der Markt dafür lag 2025 bei rund <strong>700 Millionen Euro</strong> und wächst weiter. Wenn diese Produkte das Problem lösen würden, müsste die Zahl der Betroffenen sinken. <em>Sie steigt.</em></p>

        <p class="bslp-story-statement">Diese Produkte machen deinen Alltag erträglicher. Sie ändern nichts an der Ursache. Wenn dir das reichen würde, würdest du diese Seite nicht lesen.</p>

  <!--
    Rendert bis zu 4 Testimonial-Cards aus BSLP_CONFIG.testimonials.
    Jedes Element: { name, context, youtubeId, quote }.
    Klick auf das Thumbnail lädt YouTube-nocookie iframe (privacy-friendly + schnell).
    Hinweis: Falls Error 153 erscheint, bedeutet das, dass der YouTube-Channel
    Embedding nur für bestimmte Domains erlaubt. Auf barbarascharrer.de (Webflow)
    sollte es funktionieren – im lokalen file://-Preview ggf. nicht.
  -->
  <script>
    (function(){
      /* ---------- 1) Testimonial-Cards aus BSLP_CONFIG rendern ---------- */
      function esc(s){
        return String(s == null ? '' : s)
          .replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
          .replace(/"/g,'&quot;').replace(/'/g,'&#39;');
      }
      var starSvg = '<svg viewBox="0 0 24 24" fill="currentColor"><path d="M12 17.27L18.18 21l-1.64-7.03L22 9.24l-7.19-.61L12 2 9.19 8.63 2 9.24l5.46 4.73L5.82 21z"/></svg>';
      var starsHtml = '<span class="bslp-test-stars" aria-label="5 von 5 Sternen">' + starSvg + starSvg + starSvg + starSvg + starSvg + '</span>';

      function extractYtId(input) {
        var s = (input || '').toString().trim();
        if (!s) return '';
        var m = s.match(/(?:youtu\.be\/|v=|\/embed\/|\/shorts\/)([A-Za-z0-9_-]{6,})/);
        if (m) return m[1];
        if (/^[A-Za-z0-9_-]{6,}$/.test(s)) return s;
        return '';
      }

      function renderCard(t){
        if (!t) return '';
        var name    = esc(t.name || '');
        var context = esc(t.context || '');
        var quote   = esc(t.quote || '');
        var ytId    = extractYtId(t.youtubeId);
        var thumb   = ytId ? ('https://i.ytimg.com/vi/' + encodeURIComponent(ytId) + '/hqdefault.jpg') : '';
        var videoBlock = ytId
          ? '<div class="bslp-test-video" data-yt="' + esc(ytId) + '">' +
              '<button class="bslp-test-play" type="button" aria-label="Video von ' + name + ' abspielen">' +
                '<img class="bslp-test-thumb" src="' + esc(thumb) + '" alt="" loading="lazy" />' +
                '<span class="bslp-test-play-icon" aria-hidden="true">' +
                  '<svg viewBox="0 0 24 24" fill="currentColor"><path d="M8 5v14l11-7z"/></svg>' +
                '</span>' +
              '</button>' +
            '</div>'
          : '';
        return '<article class="bslp-test-card">' +
                 videoBlock +
                 '<div class="bslp-test-body">' +
                   '<div class="bslp-test-meta">' +
                     '<span class="bslp-test-meta-name">' + name + '</span>' +
                     (context ? '<span class="bslp-test-meta-context">' + context + '</span>' : '') +
                     starsHtml +
                   '</div>' +
                   (quote ? '<p class="bslp-test-quote">' + quote + '</p>' : '') +
                 '</div>' +
               '</article>';
      }

      function renderTestimonials(){
        var grid = document.querySelector('[data-bslp-testimonials-grid]');
        if (!grid) return;
        var list = (window.BSLP_CONFIG && Array.isArray(window.BSLP_CONFIG.testimonials))
          ? window.BSLP_CONFIG.testimonials.slice(0, 4)
          : [];
        if (!list.length) { grid.innerHTML = ''; return; }
        grid.innerHTML = list.map(renderCard).join('');
      }
      renderTestimonials();

      /* ---------- 2) Lite-YT: Iframe erst beim Klick laden ---------- */
      document.addEventListener('click', function(e){
        var btn = e.target.closest('.bslp-test-play');
        if (!btn) return;
        var wrap = btn.closest('.bslp-test-video');
        if (!wrap || !wrap.dataset.yt) return;
        e.preventDefault();
        var id = wrap.dataset.yt;
        var iframe = document.createElement('iframe');
        iframe.src = 'https://www.youtube-nocookie.com/embed/' + id +
                     '?autoplay=1&rel=0&modestbranding=1&playsinline=1&feature=oembed';
        iframe.title = 'Kundenstimme Video';
        iframe.allow = 'accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share';
        iframe.allowFullscreen = true;
        iframe.referrerPolicy = 'strict-origin-when-cross-origin';
        iframe.style.cssText = 'width:100%;height:100%;border:0;display:block;';
        wrap.innerHTML = '';
        wrap.appendChild(iframe);
      }, false);

      /* ---------- 3) Hero-Video: Iframe erst beim Klick laden ---------- */
      document.addEventListener('click', function(e){
        var btn = e.target.closest('.bslp-hero-video-play');
        if (!btn) return;
        var wrap = btn.closest('[data-bslp-hero-video]');
        if (!wrap || !wrap.dataset.yt) return;
        e.preventDefault();
        var id = wrap.dataset.yt;
        var iframe = document.createElement('iframe');
        iframe.src = 'https://www.youtube-nocookie.com/embed/' + id +
                     '?autoplay=1&rel=0&modestbranding=1&playsinline=1&feature=oembed';
        iframe.title = 'Video von Barbara Scharrer';
        iframe.allow = 'accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share';
        iframe.allowFullscreen = true;
        iframe.referrerPolicy = 'strict-origin-when-cross-origin';
        iframe.style.cssText = 'width:100%;height:100%;border:0;display:block;';
        wrap.innerHTML = '';
        wrap.appendChild(iframe);
      }, false);
    })();
  </script>

            function validatePhoneNumber(prefix, rawValue) {
              var p = (prefix || "").trim();
              var raw = (rawValue || "").trim();
              if (!raw) return null;

              var cleaned = raw.replace(/[\s\-\/\.\(\)]/g, "");
              var alreadyIntl = cleaned.charAt(0) === "+";
              if (alreadyIntl) cleaned = cleaned.slice(1);

              if (!/^\d+$/.test(cleaned))
                return "Bitte gib deine Nummer nur mit Ziffern, Leerzeichen oder Bindestrichen ein.";
              if (cleaned.length < 6 || cleaned.length > 15)
                return "Bitte gib eine vollständige Telefonnummer ein (6–15 Ziffern).";

              if (!alreadyIntl && p && cleaned.charAt(0) === "0")
                return "Bitte ohne führende 0 nach der Ländervorwahl (z. B. " + p + " 176 12345678).";

              if (/(\d)\1{4,}/.test(cleaned))
                return "Bitte gib deine echte Telefonnummer ein.";

              if (/^(\d)\1+$/.test(cleaned))
                return "Bitte gib deine echte Telefonnummer ein.";

              var asc = true, desc = true;
              for (var i = 1; i < cleaned.length; i++) {
                if (+cleaned[i] !== +cleaned[i - 1] + 1) asc = false;
                if (+cleaned[i] !== +cleaned[i - 1] - 1) desc = false;
              }
              if (asc || desc) return "Bitte gib deine echte Telefonnummer ein.";

              var maxBlock = Math.min(4, Math.floor(cleaned.length / 2));
              for (var bl = 2; bl <= maxBlock; bl++) {
                var blk = cleaned.slice(0, bl);
                var rep = "";
                while (rep.length < cleaned.length) rep += blk;
                if (rep.slice(0, cleaned.length) === cleaned)
                  return "Bitte gib deine echte Telefonnummer ein.";
              }

              var uniq = {};
              for (var j = 0; j < cleaned.length; j++) uniq[cleaned[j]] = 1;
              if (Object.keys(uniq).length < 3)
                return "Bitte gib deine echte Telefonnummer ein.";

              var zeros = (cleaned.match(/0/g) || []).length;
              if (zeros / cleaned.length > 0.5)
                return "Bitte gib deine echte Telefonnummer ein.";

              var lp = window.libphonenumber;
              if (lp && p) {
                try {
                  var full = alreadyIntl ? "+" + cleaned : p + cleaned;
                  var parser = lp.parsePhoneNumberFromString || lp.parsePhoneNumber;
                  if (parser) {
                    var parsed = parser(full);
                    if (parsed && typeof parsed.isPossible === "function" && !parsed.isPossible()) {
                      return "Diese Nummer scheint nicht vollständig zu sein. Bitte prüfen.";
                    }
                  }
                } catch (e) {
                  /* Library-Fehler nicht propagieren — die Heuristiken haben bereits gefiltert. */
                }
              }

              return null;
            }

                  // und die Nummer nicht selbst international beginnt. Bleibt
                  // mind. eine Nicht-Null-Ziffer übrig, damit das Feld nicht
                  // bei Nur-Null-Eingaben leer wird.
                  var prefixEl = $id("phone_prefix");
                  var prefixVal = prefixEl ? prefixEl.value : "";
                  if (prefixVal && val.charAt(0) === "0" && /[1-9]/.test(val)) {
                    var stripped = val.replace(/^0+/, "");
                    if (stripped && stripped !== val) {
                      val = stripped;
                      this.value = stripped;
                    }
                  }
                  var err = validatePhoneNumber(prefixVal, val);
                  if (err) setFieldError("phone", err);
                  else clearFieldError("phone");
                });
              }
            }

            if (thumbEl.src.indexOf('maxresdefault') !== -1) {
              thumbEl.src = 'https://i.ytimg.com/vi/' + encodeURIComponent(heroYt) + '/hqdefault.jpg';
            }
          };
        }
      } else {
        /* Kein Video konfiguriert → Hero-Video-Block ausblenden. */
        heroWrap.style.display = 'none';
      }
    }

    /* Form-Hidden-Felder: Datum + Uhrzeit aus BSLP_CONFIG übernehmen */
    var fdate = document.querySelector('#form-webinar [name="datum"]');
    var ftime = document.querySelector('#form-webinar [name="uhrzeit"]');
    if (fdate && c.date) fdate.value = c.date;
    if (ftime && c.time) ftime.value = c.time;
  })();
</script>
