<script>
  (function(){
    /* Content wird direkt aus dem GitHub-Repo geladen.
       raw.githubusercontent.com cached nur 5 Minuten — nach git push
       ist die neue Version in max. 5 Min live. */
    var CONTENT_URL = 'https://raw.githubusercontent.com/privatdaniel/barbara-scharrer-website/main/webinar-danke-content.txt';

    var mount = document.getElementById('bsty-mount');
    if (!mount) return;

    function showError(msg, detail) {
      mount.innerHTML =
        '<div style="padding:2rem 1.5rem;text-align:center;color:#7b2020;background:#fff5f5;border:1px solid #d9534f;border-radius:8px;font-family:system-ui,-apple-system,Segoe UI,Roboto,sans-serif;font-size:14px;line-height:1.5">' +
          '<strong style="font-size:15px">Danke-Seite konnte nicht geladen werden.</strong><br>' +
          '<span style="opacity:0.8">' + msg + '</span>' +
          (detail ? '<br><small style="opacity:0.6;font-family:monospace;font-size:12px">' + detail + '</small>' : '') +
        '</div>';
    }

    if (!window.fetch) { showError('Browser unterstützt fetch() nicht.', ''); return; }

    fetch(CONTENT_URL, { credentials: 'omit', cache: 'no-cache' })
      .then(function(r){
        if (!r.ok) throw new Error('HTTP ' + r.status + ' ' + r.statusText);
        return r.text();
      })
      .then(function(html){
        if (!html || html.length < 100) throw new Error('Empty response');
        mount.innerHTML = html;

        /* <script>-Tags aus innerHTML manuell re-aktivieren */
        var scripts = mount.querySelectorAll('script');
        scripts.forEach(function(oldScript){
          var newScript = document.createElement('script');
          for (var i = 0; i < oldScript.attributes.length; i++) {
            var a = oldScript.attributes[i];
            newScript.setAttribute(a.name, a.value);
          }
          newScript.textContent = oldScript.textContent;
          oldScript.parentNode.replaceChild(newScript, oldScript);
        });

        console.info('[Danke-Seite] geladen ('+ html.length +' Bytes, '+ scripts.length +' Scripts)');
      })
      .catch(function(err){
        console.error('[Danke-Seite] Content-Load fehlgeschlagen:', err);
        showError(err.message || 'Unbekannter Fehler', CONTENT_URL);
      });
  })();
</script>
